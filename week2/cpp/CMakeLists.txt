cmake_minimum_required(VERSION 3.16)
project(comento_week2_cpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(COMENTO_ENABLE_GOOGLETEST "Compile real GoogleTest/gMock tests" ON)
option(COMENTO_FETCH_GOOGLETEST "Fetch pinned GoogleTest v1.18.0 when not installed (needs network)" OFF)
option(COMENTO_REQUIRE_OPENCV_FOR_GTEST "Require OpenCV tests when GoogleTest is enabled" ON)

add_library(depth_projection STATIC src/depth_projection.cpp)
target_include_directories(depth_projection PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
    target_compile_options(depth_projection PRIVATE /W4)
else()
    target_compile_options(depth_projection PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()

enable_testing()
add_executable(cpp_core_smoke tests/core_smoke.cpp)
target_link_libraries(cpp_core_smoke PRIVATE depth_projection)
add_test(NAME cpp_core_smoke COMMAND cpp_core_smoke)

find_package(OpenCV QUIET COMPONENTS core imgproc imgcodecs)
if(OpenCV_FOUND)
    add_library(opencv_depth_source STATIC src/opencv_depth_source.cpp)
    target_link_libraries(opencv_depth_source PUBLIC depth_projection ${OpenCV_LIBS})
    target_include_directories(opencv_depth_source PUBLIC ${OpenCV_INCLUDE_DIRS} ${CMAKE_CURRENT_SOURCE_DIR}/include)
    add_executable(cpp_depth_demo src/depth_demo.cpp)
    target_link_libraries(cpp_depth_demo PRIVATE opencv_depth_source)
    # Test-only bridge for actual C++/Python differential checks.
    add_executable(cpp_xyz_probe tests/xyz_probe.cpp)
    target_link_libraries(cpp_xyz_probe PRIVATE opencv_depth_source)
else()
    message(STATUS "OpenCV C++ dev headers not found: adapter and OpenCV unit tests unavailable")
endif()

if(COMENTO_ENABLE_GOOGLETEST)
    if(COMENTO_REQUIRE_OPENCV_FOR_GTEST AND NOT OpenCV_FOUND)
        message(FATAL_ERROR "OpenCV C++ development package missing: full GoogleTest suite requires OpenCV. Install libopencv-dev (Ubuntu) before configuring.")
    endif()
    find_package(GTest CONFIG QUIET)
    if(NOT GTest_FOUND AND COMENTO_FETCH_GOOGLETEST)
        include(FetchContent)
        FetchContent_Declare(googletest
            GIT_REPOSITORY https://github.com/google/googletest.git
            GIT_TAG v1.18.0)
        FetchContent_MakeAvailable(googletest)
    elseif(NOT GTest_FOUND)
        message(FATAL_ERROR "GoogleTest/gMock not found. Install both packages, or pass -DCOMENTO_FETCH_GOOGLETEST=ON with internet. Offline core smoke: -DCOMENTO_ENABLE_GOOGLETEST=OFF")
    endif()
    add_executable(cpp_gtests tests/test_depth_projection.cpp)
    # gmock_main initializes gMock flags as well as GoogleTest flags.
    target_link_libraries(cpp_gtests PRIVATE depth_projection GTest::gmock_main GTest::gmock)
    if(OpenCV_FOUND)
        target_sources(cpp_gtests PRIVATE tests/test_opencv_adapter.cpp)
        target_link_libraries(cpp_gtests PRIVATE opencv_depth_source)
    endif()
    include(GoogleTest)
    gtest_discover_tests(cpp_gtests)
endif()
